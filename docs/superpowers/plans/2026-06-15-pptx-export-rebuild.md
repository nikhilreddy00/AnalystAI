"""
Gap 4 — PNG chart renderer for the PowerPoint export.

Public entry: render_chart_to_png(config) -> bytes | None.

Uses matplotlib (+ seaborn for heatmaps) so the deck preserves the web
dashboard's semantic coloring (e.g., survival bars green/red, not all blue).
matplotlib is forced to the Agg backend so it works headless on the server.
matplotlib + seaborn are imported lazily inside the render entry point so
this module stays cheap to import on non-PPTX cold paths.
"""
from __future__ import annotations

import warnings
from io import BytesIO

from orchestrator.state import ChartConfig, ChartType

# Brand accent — matches the KPI shape border in report_pptx.py.
DEFAULT_COLOR = "#534AB7"

# Quiet seaborn's DeprecationWarning noise on certain pandas combos.
warnings.filterwarnings("ignore", category=DeprecationWarning, module="seaborn")


def render_chart_to_png(config: ChartConfig) -> bytes | None:
    """Render a ChartConfig to PNG bytes via matplotlib.

    Returns None for unsupported chart types or on render exception —
    caller falls back to a data table inside the chart slide. The deck
    never errors out on a single bad chart.
    """
    # Lazy imports: matplotlib has ~300ms cold-start; we don't pay it on
    # routes that never call render_chart_to_png.
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt  # noqa: F401  (imported for side effects + downstream use)

    try:
        return _dispatch(config, plt)
    except Exception:
        return None


def _dispatch(config: ChartConfig, plt) -> bytes | None:
    """Type-dispatch. Returns None for unhandled types."""
    # Stubs filled in by Tasks 2 + 3.
    return None


def _resolve_colors(config: ChartConfig, n: int) -> list[str]:
    """If config.colors is populated, cycle it to length n; else DEFAULT_COLOR repeated."""
    if not config.colors:
        return [DEFAULT_COLOR] * n
    return [config.colors[i % len(config.colors)] for i in range(n)]


def _finalize_axes(ax, value_format: str = "number") -> None:
    """Apply the look shared across all chart types: hidden spines, font sizes,
    and percent-formatted y-axis when applicable."""
    ax.spines["top"].set_visible(False)
    ax.spines["right"].set_visible(False)
    ax.tick_params(axis="both", labelsize=10)
    if hasattr(ax, "xaxis"):
        ax.xaxis.label.set_size(11)
    if hasattr(ax, "yaxis"):
        ax.yaxis.label.set_size(11)
    if value_format == "percent":
        import matplotlib.pyplot as plt
        ax.yaxis.set_major_formatter(plt.FuncFormatter(lambda x, _: f"{x:.0f}%"))


def _fig_to_png(fig) -> bytes:
    """Save the figure to PNG bytes and close it to free memory."""
    import matplotlib.pyplot as plt
    buf = BytesIO()
    fig.savefig(buf, format="png", dpi=150, bbox_inches="tight")
    plt.close(fig)
    return buf.getvalue()
```

def _render_bar(config: ChartConfig, plt) -> bytes | None:
    """Vertical bar chart with value labels on top of each bar."""
    if not config.data or not config.y_keys:
        return None
    x_key = config.x_key
    y_key = config.y_keys[0]
    labels = [str(r.get(x_key, "")) for r in config.data]
    values = [_safe_float(r.get(y_key)) for r in config.data]
    colors = _resolve_colors(config, len(values))

    fig, ax = plt.subplots(figsize=(10, 5.5), dpi=150)
    bars = ax.bar(labels, values, color=colors)

    fmt = "{:.1f}%" if config.value_format == "percent" else "{:g}"
    for bar, v in zip(bars, values):
        ax.text(
            bar.get_x() + bar.get_width() / 2,
            bar.get_height(),
            fmt.format(v),
            ha="center",
            va="bottom",
            fontsize=10,
            color="#333333",
        )

    if config.x_label:
        ax.set_xlabel(config.x_label)
    if config.y_label:
        ax.set_ylabel(config.y_label)
    _finalize_axes(ax, config.value_format)
    return _fig_to_png(fig)


def _safe_float(v) -> float:
    try:
        return float(v) if v is not None else 0.0
    except (TypeError, ValueError):
        return 0.0
```

def _render_line(config: ChartConfig, plt) -> bytes | None:

        ax.text(

def _render_histogram(config: ChartConfig, plt) -> bytes | None:
    """Histogram. When multiple y_keys are present (target-split case),
    draws overlapping bars with alpha=0.7 and a legend."""
    if not config.data or not config.y_keys:
        return None
    xs = [_safe_float(r.get(config.x_key)) for r in config.data]
    colors = _resolve_colors(config, len(config.y_keys))

    fig, ax = plt.subplots(figsize=(10, 5.5), dpi=150)
    # Width = ~85% of the bin spacing for visual breathing room.
    width = (xs[1] - xs[0]) * 0.85 if len(xs) >= 2 else 1.0

    multi = len(config.y_keys) > 1
    for i, y_key in enumerate(config.y_keys):
        ys = [_safe_float(r.get(y_key)) for r in config.data]
        ax.bar(
            xs, ys,
            width=width,
            color=colors[i],
            alpha=0.7 if multi else 1.0,
            label=y_key,
            edgecolor="none",
        )

    if multi and config.show_legend:
        ax.legend(loc="best", fontsize=10, frameon=False)

    if config.x_label:
        ax.set_xlabel(config.x_label)
    if config.y_label:
        ax.set_ylabel(config.y_label)
    _finalize_axes(ax, config.value_format)
    return _fig_to_png(fig)
```

def _render_heatmap(config: ChartConfig, plt) -> bytes | None:
    """Correlation-style heatmap via seaborn. Data is wide-form:
    [{<x_key>: row_label, col1: val, col2: val, ...}, ...].
    Row labels = x_key values; column labels = config.y_keys."""
    if not config.data or not config.y_keys:
        return None

    import seaborn as sns
    import numpy as np

    row_labels = [str(r.get(config.x_key, "")) for r in config.data]
    col_labels = list(config.y_keys)
    matrix = np.array(
        [[_safe_float(r.get(col)) for col in col_labels] for r in config.data],
        dtype=float,
    )

    fig, ax = plt.subplots(figsize=(10, 5.5), dpi=150)
    sns.heatmap(
        matrix,
        annot=True,
        fmt=".2f",
        cmap="RdBu_r",
        center=0,
        xticklabels=col_labels,
        yticklabels=row_labels,
        ax=ax,
        cbar=True,
        annot_kws={"fontsize": 9},
    )
    plt.setp(ax.get_xticklabels(), rotation=30, ha="right", fontsize=10)
    plt.setp(ax.get_yticklabels(), rotation=0, fontsize=10)
    return _fig_to_png(fig)
```

def _add_kpi_slide(prs: Presentation, session: AnalysisSession) -> None:
    kpis = session.analysis_findings.kpis if session.analysis_findings else []
    kpis = (kpis or [])[:5]
    if not kpis:
        return  # Skip slide entirely when no KPIs.

    slide = prs.slides.add_slide(prs.slide_layouts[6])
    _slide_title(slide, "Key Metrics")

    n = len(kpis)
    gap = Inches(0.3)
    total_gap = gap * (n - 1) if n > 1 else 0
    shape_w = Inches((CONTENT_W - total_gap) / Inches(1) / n)
    shape_h = Inches(1.8)
    top = Inches(1.8)
    left = MARGIN

    for i, kpi in enumerate(kpis):
        cur_left = Inches(MARGIN / Inches(1) + i * (shape_w / Inches(1) + gap / Inches(1)))
        _add_kpi_shape(slide, kpi, cur_left, top, shape_w, shape_h)


def _add_kpi_shape(slide, kpi: KPI, left, top, width, height) -> None:
    shape = slide.shapes.add_shape(MSO_SHAPE.ROUNDED_RECTANGLE, left, top, width, height)
    shape.fill.solid()
    shape.fill.fore_color.rgb = KPI_BG
    shape.line.color.rgb = ACCENT_PURPLE
    shape.line.width = Pt(1)
    shape.shadow.inherit = False

    tf = shape.text_frame
    tf.word_wrap = True
    tf.vertical_anchor = MSO_ANCHOR.TOP
    tf.margin_left = Inches(0.15)
    tf.margin_right = Inches(0.15)
    tf.margin_top = Inches(0.18)
    tf.margin_bottom = Inches(0.15)

    # Label
    p_label = tf.paragraphs[0]
    p_label.alignment = PP_ALIGN.CENTER
    _set_run(p_label.add_run(), kpi.label, size=Pt(11), color=CAPTION_GRAY, font_name=BODY_FONT)

    # Value (big number)
    p_val = tf.add_paragraph()
    p_val.alignment = PP_ALIGN.CENTER
    p_val.space_before = Pt(6)
    _set_run(p_val.add_run(), kpi.value, size=Pt(36), bold=True, color=NEAR_BLACK, font_name=BODY_FONT)

    # Context
    if kpi.context:
        p_ctx = tf.add_paragraph()
        p_ctx.alignment = PP_ALIGN.CENTER
        p_ctx.space_before = Pt(6)
        _set_run(p_ctx.add_run(), kpi.context, size=Pt(10), color=CAPTION_GRAY, font_name=BODY_FONT)


def _add_summary_slide(prs: Presentation, session: AnalysisSession) -> None:
    bullets = _exec_summary_bullets(session)
    if not bullets:
        return
    slide = prs.slides.add_slide(prs.slide_layouts[6])
    _slide_title(slide, "Executive Summary")

    tb = slide.shapes.add_textbox(MARGIN, Inches(1.5), CONTENT_W, Inches(5.0))
    tf = tb.text_frame
    tf.word_wrap = True

    for i, bullet in enumerate(bullets[:5]):
        p = tf.paragraphs[0] if i == 0 else tf.add_paragraph()
        p.alignment = PP_ALIGN.LEFT
        p.space_after = Pt(12)
        # Bullet marker
        _set_run(p.add_run(), "•  ", size=Pt(16), color=BODY_GRAY, font_name=BODY_FONT)
        # Bullet text with numbers bolded
        for piece, is_num in _split_bold_numbers(bullet):
            _set_run(p.add_run(), piece, size=Pt(16), bold=is_num, color=BODY_GRAY, font_name=BODY_FONT)


def _add_chart_slide(prs: Presentation, chart: ChartConfig) -> None:
    """Render one chart slide. Image preferred, table fallback when unrenderable."""
    slide = prs.slides.add_slide(prs.slide_layouts[6])
    _slide_title(slide, chart.title, size=Pt(24))

    # Lazy import to keep the cold-start path of report_pptx light.

    png_bytes = render_chart_to_png(chart)

    if png_bytes:
        slide.shapes.add_picture(
            BytesIO(png_bytes),
            Inches(1.33), Inches(1.4),

# ── Helpers ───────────────────────────────────────────────────────────────────

def _exec_summary_bullets(session: AnalysisSession) -> list[str]:
    """Source priority: executive_view.findings → split executive_summary into sentences."""
    if session.insight_report is None:
        return []
    ev = session.insight_report.executive_view
    if ev and ev.findings:
        return [s for s in ev.findings if s.strip()][:5]
    summary = session.insight_report.executive_summary or ""
    if not summary:
        return []
    # Split on sentence boundaries; keep up to 5.
    parts = re.split(r"(?<=[.!?])\s+(?=[A-Z])", summary.strip())
    return [p.strip() for p in parts if p.strip()][:5]


def _action_items(session: AnalysisSession) -> list[tuple[str, str]]:
    """Returns (action, rationale) tuples. Source priority:
    executive_view.action_items (action + expected_impact) →
    insight_report.recommended_actions (action + reasoning)."""
    if session.insight_report is None:
        return []
    ev = session.insight_report.executive_view
    if ev and ev.action_items:
        return [(a.action, a.expected_impact) for a in ev.action_items[:4] if a.action]
    legacy: list[RecommendedAction] = session.insight_report.recommended_actions or []
    return [(a.action, a.reasoning) for a in legacy[:4] if a.action]


def _split_bold_numbers(text: str) -> Iterable[tuple[str, bool]]:
    """Yield (piece, is_number) tuples by splitting on NUMBER_RE matches.
    The number tokens get rendered bold; everything else stays regular."""
    pos = 0
    for m in NUMBER_RE.finditer(text):
        if m.start() > pos:
            yield text[pos:m.start()], False
        yield m.group(0), True
        pos = m.end()
    if pos < len(text):
        yield text[pos:], False


def _slide_title(slide, text: str, size=Pt(28)) -> None:
    """Standard slide title. Georgia bold, NEAR_BLACK, left-aligned, top.
    No accent rule (the legacy indigo rectangle is removed in Gap 4)."""
    _textbox(
        slide, left=MARGIN, top=Inches(0.5), width=CONTENT_W, height=Inches(0.8),
        text=text,
        font_size=size, bold=True, color=NEAR_BLACK, font_name=TITLE_FONT,
        align=PP_ALIGN.LEFT,
    )


def _textbox(
    slide, left, top, width, height,
    text: str, font_size, bold: bool = False, italic: bool = False,
    color: RGBColor | None = None, font_name: str = BODY_FONT,
    align=PP_ALIGN.LEFT,
):
    tb = slide.shapes.add_textbox(left, top, width, height)
    tf = tb.text_frame
    tf.word_wrap = True
    p = tf.paragraphs[0]
    p.alignment = align
    _set_run(p.add_run(), text, size=font_size, bold=bold, italic=italic, color=color, font_name=font_name)
    return tb


def _set_run(run, text: str, *, size, bold: bool = False, italic: bool = False,
             color: RGBColor | None = None, font_name: str = BODY_FONT) -> None:
    run.text = text
    run.font.size = size
    run.font.bold = bold
    run.font.italic = italic
    run.font.name = font_name
    if color is not None:
        run.font.color.rgb = color


def _set_background(slide, color: RGBColor) -> None:
    bg = slide.background
    fill = bg.fill
    fill.solid()
    fill.fore_color.rgb = color


def _add_table(slide, columns: list[str], rows: list[dict], left, top, width, height):
    """Plain data table used only as a chart-slide fallback."""
    if not rows:
        return None
    n_rows = len(rows) + 1
    n_cols = len(columns)
    table_shape = slide.shapes.add_table(n_rows, n_cols, left, top, width, height)
    tbl = table_shape.table

    for j, col in enumerate(columns):
        cell = tbl.cell(0, j)
        cell.text = str(col)
        for p in cell.text_frame.paragraphs:
            for run in p.runs:
                run.font.bold = True
                run.font.size = Pt(11)
                run.font.name = BODY_FONT

    for i, row in enumerate(rows, start=1):
        for j, col in enumerate(columns):
            cell = tbl.cell(i, j)
            cell.text = _fmt(row.get(col, ""))
            for p in cell.text_frame.paragraphs:
                for run in p.runs:
                    run.font.size = Pt(10)
                    run.font.name = BODY_FONT
    return table_shape


def _fmt(v: Any) -> str:
    if v is None:
        return "—"
    if isinstance(v, float):
        if v != v:
            return "—"
        return f"{v:.2f}" if not v.is_integer() else str(int(v))
    return str(v)
